cmake_minimum_required(VERSION 3.20)
option(WG_DEVICE "Build official DISPLAY SD application" OFF)
set(WILIBSP_PATH "${CMAKE_CURRENT_SOURCE_DIR}/wilibsp" CACHE PATH "Pinned official BSP")
set(ONEWILI_PATH "${CMAKE_CURRENT_SOURCE_DIR}/onewili" CACHE PATH "Pinned OneWili b0eeccda21b0594c8062cd17e9755f0c26b0cf6b")
if(WG_DEVICE)
  set(PICO_BOARD freewili2 CACHE STRING "Board")
  list(APPEND PICO_BOARD_HEADER_DIRS "${WILIBSP_PATH}/bsp/boards")
  set(PICO_PIO_USB_PATH "${WILIBSP_PATH}/bsp/third_party/Pico-PIO-USB")
  include("${WILIBSP_PATH}/pico_sdk_import.cmake")
  project(WiliGlitch C CXX ASM)
  find_package(Python3 COMPONENTS Interpreter REQUIRED)
  set(FW2_AGENTIO OFF CACHE BOOL "Offline app" FORCE)
  pico_sdk_init()
  add_subdirectory("${WILIBSP_PATH}/bsp" "${CMAKE_BINARY_DIR}/bsp")
  add_subdirectory("${ONEWILI_PATH}/wilibsp" "${CMAKE_BINARY_DIR}/onewili")
  # Official supported tunables, shared by library and consumers. The only
  # queried payload is a 16-byte OTP chunk; stock 4K response buffers would
  # exceed the SDK's scratch-bank stack during nested generated calls.
  target_compile_definitions(onewili_fwgui PUBLIC
    OW_CMD_MAX=128 OW_RESP_MAX=512 OW_TEXT_EVENT_QUEUE=4 OW_DEFAULT_TIMEOUT_MS=1000)
  target_compile_options(onewili_fwgui PRIVATE -fstack-usage)
else()
  project(WiliGlitch C)
endif()
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
add_library(wg_model src/model.c src/adapter.c src/hardware.c)
target_include_directories(wg_model PUBLIC src)
target_compile_options(wg_model PRIVATE -Wall -Wextra -Werror)
if(WG_DEVICE)
  add_executable(WiliGlitch src/main.c src/ui.c src/api_contract.c src/onewili_adapter.c)
  target_compile_definitions(WiliGlitch PRIVATE PICO_STACK_SIZE=4096)
  target_include_directories(WiliGlitch PRIVATE "${ONEWILI_PATH}/wilibsp/include")
  target_link_libraries(WiliGlitch PRIVATE wg_model freewili2_bsp onewili_fwgui)
  set_source_files_properties(src/main.c src/ui.c src/api_contract.c src/onewili_adapter.c PROPERTIES COMPILE_OPTIONS "-Wall;-Wextra;-Werror")
  pico_enable_stdio_usb(WiliGlitch 0)
  pico_enable_stdio_uart(WiliGlitch 0)
  fw2_display_app(WiliGlitch NAME WiliGlitch VERSION 003
    DESCRIPTION "Passive UI regression diagnostic; OneWili startup/retries paused"
    POWER_ZONES DISPLAY)
else()
  enable_testing()
  add_executable(wg_tests tests/test_model.c)
  target_link_libraries(wg_tests PRIVATE wg_model)
  target_compile_options(wg_tests PRIVATE -Wall -Wextra -Werror -UNDEBUG)
  add_test(NAME foundation COMMAND wg_tests)
  add_executable(wg_hardware_tests tests/test_hardware.c)
  target_link_libraries(wg_hardware_tests PRIVATE wg_model)
  target_compile_options(wg_hardware_tests PRIVATE -Wall -Wextra -Werror -UNDEBUG)
  add_test(NAME hardware_adapter COMMAND wg_hardware_tests)
  if(EXISTS "${ONEWILI_PATH}/wilibsp/src/onewili.c")
    add_executable(wg_onewili_tests tests/test_onewili.c "${ONEWILI_PATH}/wilibsp/src/onewili.c")
    target_include_directories(wg_onewili_tests PRIVATE "${ONEWILI_PATH}/wilibsp/include")
    target_compile_definitions(wg_onewili_tests PRIVATE OW_CMD_MAX=128 OW_RESP_MAX=512
      OW_TEXT_EVENT_QUEUE=4 OW_DEFAULT_TIMEOUT_MS=1000)
    target_compile_options(wg_onewili_tests PRIVATE -UNDEBUG)
    add_test(NAME official_api_transport COMMAND wg_onewili_tests)
  endif()
  if(EXISTS "${WILIBSP_PATH}/bsp/display/st7796.h")
    add_executable(wg_home_recovery_tests "${WILIBSP_PATH}/tests/test_app_recovery.c"
      "${WILIBSP_PATH}/bsp/input/app_recovery.c")
    target_include_directories(wg_home_recovery_tests PRIVATE "${WILIBSP_PATH}/bsp")
    target_compile_definitions(wg_home_recovery_tests PRIVATE HOST_TEST)
    target_compile_options(wg_home_recovery_tests PRIVATE -UNDEBUG)
    add_test(NAME official_home_recovery COMMAND wg_home_recovery_tests)
    add_executable(wg_render_tests tests/test_render.c src/ui.c)
    target_include_directories(wg_render_tests PRIVATE "${WILIBSP_PATH}/bsp")
    target_link_libraries(wg_render_tests PRIVATE wg_model)
    target_compile_options(wg_render_tests PRIVATE -Wall -Wextra -Werror -UNDEBUG)
    add_test(NAME rendering COMMAND wg_render_tests)
  endif()
endif()
